+++
title = "Cut Paste"
+++

- <kbd>Ctrl</kbd> + <kbd>w</kbd>: Cut the word before the cursor
- <kbd>Ctrl</kbd> + <kbd>u</kbd>: Cut the line before the cursor
- <kbd>Ctrl</kbd> + <kbd>k</kbd>: Cut the line after the cursor
- <kbd>Ctrl</kbd> + <kbd>y</kbd>: Paste (yank) from the terminal's clipboard
- <kbd>Alt</kbd> + <kbd>.</kbd>: Recall the last arguments of the previous command
